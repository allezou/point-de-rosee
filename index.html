  <!-- Script des calculs -->

  <script>

    // Calcul du point de rosée (formule de Magnus-Tetens)

    function calculateDewPoint(temp, rh) {

      if (isNaN(temp) || isNaN(rh) || rh <= 0) return null;

      const a = 17.27;

      const b = 237.7;

      const alpha = ((a * temp) / (b + temp)) + Math.log(rh / 100);

      const dewPoint = (b * alpha) / (a - alpha);

      return dewPoint;

    }


    function updateCalculations() {

      const tempIn = parseFloat(document.getElementById('temp-in').value);

      const humIn = parseFloat(document.getElementById('hum-in').value);

      const tempOut = parseFloat(document.getElementById('temp-out').value);

      const humOut = parseFloat(document.getElementById('hum-out').value);


      const dpIn = calculateDewPoint(tempIn, humIn);

      const dpOut = calculateDewPoint(tempOut, humOut);


      const dpInEl = document.getElementById('dp-in');

      const dpOutEl = document.getElementById('dp-out');

      const card = document.getElementById('result-card');

      const icon = document.getElementById('verdict-icon');

      const title = document.getElementById('verdict-title');

      const desc = document.getElementById('verdict-desc');


      if (dpIn === null || dpOut === null) {

        dpInEl.textContent = "PR: -- °C";

        dpOutEl.textContent = "PR: -- °C";

        card.className = "rounded-2xl p-5 border bg-slate-800 border-slate-700 text-slate-400 text-center";

        icon.textContent = "❓";

        title.textContent = "Données incomplètes";

        desc.textContent = "Veuillez saisir des températures et taux d'humidité valides.";

        return;

      }


      dpInEl.textContent = `PR: ${dpIn.toFixed(1)} °C`;

      dpOutEl.textContent = `PR: ${dpOut.toFixed(1)} °C`;


      const diff = dpIn - dpOut;


      if (diff > 2) {

        // OUVRIR

        card.className = "rounded-2xl p-5 border bg-emerald-950/80 border-emerald-500/50 text-emerald-100 shadow-emerald-950/50 shadow-lg text-center";

        icon.textContent = "🟢";

        title.textContent = "OUVREZ / AÉREZ";

        desc.textContent = `L'air extérieur est nettement plus sec en eau absolue (différence de ${diff.toFixed(1)} °C sur le point de rosée). Aérer accélère le séchage.`;

      } else if (diff >= -0.5 && diff <= 2) {

        // PRUDENCE

        card.className = "rounded-2xl p-5 border bg-amber-950/80 border-amber-500/50 text-amber-100 shadow-amber-950/50 shadow-lg text-center";

        icon.textContent = "🟠";

        title.textContent = "PRUDENCE";

        desc.textContent = `Les points de rosée sont très proches (écart de ${diff.toFixed(1)} °C). L'aération aura peu d'effet sur le séchage.`;

      } else {

        // NE PAS OUVRIR

        card.className = "rounded-2xl p-5 border bg-rose-950/80 border-rose-500/50 text-rose-100 shadow-rose-950/50 shadow-lg text-center";

        icon.textContent = "🔴";

        title.textContent = "NE PAS AÉRER";

        desc.textContent = `L'air extérieur apporterait de l'humidité supplémentaire dans votre pièce (Point de rosée ext. plus élevé de ${Math.abs(diff).toFixed(1)} °C).`;

      }

    }


    // Écouteurs d'événements sur les champs de saisie

    ['temp-in', 'hum-in', 'temp-out', 'hum-out'].forEach(id => {

      document.getElementById(id).addEventListener('input', updateCalculations);

    });


    // Lancement au chargement

    updateCalculations();

  </script>
